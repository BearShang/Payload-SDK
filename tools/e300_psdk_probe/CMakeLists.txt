cmake_minimum_required(VERSION 3.16)
project(e300_psdk_probe C)
set(CMAKE_C_STANDARD 11)
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
    message(FATAL_ERROR "This probe targets the E300 ARM64 Linux environment")
endif()
set(OSAL "${SDK_ROOT}/samples/sample_c/platform/linux/common/osal")
add_executable(e300_psdk_probe main.c "${OSAL}/osal.c")
add_executable(test_uart_status test_uart_status.c "${OSAL}/osal.c")
foreach(target e300_psdk_probe test_uart_status)
    target_include_directories(${target} PRIVATE "${SDK_ROOT}/psdk_lib/include" "${OSAL}")
    target_compile_definitions(${target} PRIVATE _GNU_SOURCE)
    target_compile_options(${target} PRIVATE -Wall -Wextra)
    target_link_libraries(${target} PRIVATE
        "${SDK_ROOT}/psdk_lib/lib/aarch64-linux-gnu-gcc/libpayloadsdk.a" pthread m dl rt)
endforeach()
enable_testing()
add_test(NAME uart_status COMMAND test_uart_status)
